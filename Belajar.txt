    <script>
        let currentGroups = [];

        function handleFileUpload(event) {
            const file = event.target.files[0];
            const messageDiv = document.getElementById('message');
            
            if (!file) return;

            const reader = new FileReader();
            
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    
                    const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    const jsonData = XLSX.utils.sheet_to_json(firstSheet, { header: 1 });
                    
                    const names = jsonData
                        .map(row => row[0])
                        .filter(name => name && String(name).trim() !== '')
                        .map(name => String(name).trim());
                    
                    if (names.length === 0) {
                        messageDiv.innerHTML = '<div class="error">⚠️ File Excel tidak mengandung data yang valid!</div>';
                        return;
                    }
                    
                    document.getElementById('members').value = names.join('\n');
                    messageDiv.innerHTML = `<div class="success">✅ Berhasil import ${names.length} nama dari Excel!</div>`;
                    
                    setTimeout(() => {
                        messageDiv.innerHTML = '';
                    }, 3000);
                    
                } catch (error) {
                    messageDiv.innerHTML = '<div class="error">⚠️ Gagal membaca file Excel. Pastikan formatnya benar!</div>';
                }
            };
            
            reader.readAsArrayBuffer(file);
        }

        function randomizeGroups() {
            const membersText = document.getElementById('members').value.trim();
            const numGroups = parseInt(document.getElementById('numGroups').value);
            const maxPerGroup = parseInt(document.getElementById('maxPerGroup').value);
            const messageDiv = document.getElementById('message');
            const resultsDiv = document.getElementById('results');
            const exportBtn = document.getElementById('exportBtn');

            messageDiv.innerHTML = '';
            resultsDiv.innerHTML = '';
            currentGroups = [];
            exportBtn.disabled = true;

            if (!membersText) {
                messageDiv.innerHTML = '<div class="error">⚠️ Silakan masukkan daftar anggota!</div>';
                return;
            }

            if (numGroups < 1) {
                messageDiv.innerHTML = '<div class="error">⚠️ Jumlah kelompok harus minimal 1!</div>';
                return;
            }

            if (maxPerGroup < 1) {
                messageDiv.innerHTML = '<div class="error">⚠️ Maksimal anggota per kelompok harus minimal 1!</div>';
                return;
            }

            const members = membersText.split('\n')
                .map(m => m.trim())
                .filter(m => m.length > 0);

            if (members.length === 0) {
                messageDiv.innerHTML = '<div class="error">⚠️ Tidak ada anggota yang valid!</div>';
                return;
            }

            const totalCapacity = numGroups * maxPerGroup;
            if (members.length > totalCapacity) {
                messageDiv.innerHTML = `<div class="info">ℹ️ Total anggota (${members.length}) melebihi kapasitas (${totalCapacity}). Beberapa anggota mungkin tidak masuk kelompok.</div>`;
            }

            const shuffled = [...members].sort(() => Math.random() - 0.5);
            const groups = Array.from({ length: numGroups }, () => []);
            
            let memberIndex = 0;
            for (let i = 0; i < shuffled.length && memberIndex < shuffled.length; i++) {
                const groupIndex = i % numGroups;
                if (groups[groupIndex].length < maxPerGroup) {
                    groups[groupIndex].push(shuffled[memberIndex]);
                    memberIndex++;
                }
            }

            currentGroups = groups;

            groups.forEach((group, index) => {
                if (group.length > 0) {
                    const groupDiv = document.createElement('div');
                    groupDiv.className = 'group';
                    
                    const header = document.createElement('div');
                    header.className = 'group-header';
                    header.textContent = `Kelompok ${index + 1} (${group.length} anggota)`;
                    groupDiv.appendChild(header);

                    group.forEach(member => {
                        const memberDiv = document.createElement('div');
                        memberDiv.className = 'member';
                        memberDiv.textContent = member;
                        groupDiv.appendChild(memberDiv);
                    });

                    resultsDiv.appendChild(groupDiv);
                }
            });

            if (memberIndex < shuffled.length) {
                messageDiv.innerHTML += `<div class="error">⚠️ ${shuffled.length - memberIndex} anggota tidak dapat masuk kelompok karena keterbatasan kapasitas!</div>`;
            }

            exportBtn.disabled = false;
        }

        function exportToExcel() {
            if (currentGroups.length === 0) {
                alert('Belum ada hasil kelompok untuk di-export!');
                return;
            }

            const data = [];
            data.push(['Kelompok', 'Nama Anggota', 'No. Urut']);

            currentGroups.forEach((group, groupIndex) => {
                group.forEach((member, memberIndex) => {
                    data.push([
                        `Kelompok ${groupIndex + 1}`,
                        member,
                        memberIndex + 1
                    ]);
                });
            });

            const worksheet = XLSX.utils.aoa_to_sheet(data);
            
            const colWidths = [
                { wch: 15 },
                { wch: 30 },
                { wch: 10 }
            ];
            worksheet['!cols'] = colWidths;

            const workbook = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(workbook, worksheet, 'Hasil Kelompok');

            const timestamp = new Date().toISOString().slice(0, 10);
            XLSX.writeFile(workbook, `Hasil_Kelompok_${timestamp}.xlsx`);

            const messageDiv = document.getElementById('message');
            messageDiv.innerHTML = '<div class="success">✅ File Excel berhasil di-download!</div>';
            setTimeout(() => {
                messageDiv.innerHTML = '';
            }, 3000);
        }
    </script>
